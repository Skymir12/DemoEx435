-- ============================================

CREATE DATABASE MasterPol;
GO
USE MasterPol;
GO

-- Справочник типов партнеров
CREATE TABLE PartnerTypes (
    PartnerTypeId   INT          NOT NULL IDENTITY(1,1) PRIMARY KEY,
    PartnerTypeName NVARCHAR(50) NOT NULL UNIQUE
);
GO

-- Партнеры
CREATE TABLE Partners (
    PartnerId     INT           NOT NULL IDENTITY(1,1) PRIMARY KEY,
    PartnerTypeId INT           NOT NULL,
    PartnerName   NVARCHAR(100) NOT NULL,
    DirectorName  NVARCHAR(150) NOT NULL,
    Email         NVARCHAR(100) NOT NULL,
    Phone         NVARCHAR(20)  NOT NULL,
    LegalAddress  NVARCHAR(300) NOT NULL,
    Inn           NVARCHAR(12)  NULL,
    Rating        INT           NOT NULL CHECK (Rating >= 0),
    CONSTRAINT FK_Partners_PartnerTypes FOREIGN KEY (PartnerTypeId) REFERENCES PartnerTypes(PartnerTypeId)
);
GO

-- Типы продукции
CREATE TABLE ProductTypes (
    ProductTypeId   INT           NOT NULL IDENTITY(1,1) PRIMARY KEY,
    ProductTypeName NVARCHAR(50)  NOT NULL UNIQUE,
    Coefficient     DECIMAL(10,2) NOT NULL CHECK (Coefficient > 0)
);
GO

-- Типы материалов
CREATE TABLE MaterialTypes (
    MaterialTypeId   INT          NOT NULL IDENTITY(1,1) PRIMARY KEY,
    MaterialTypeName NVARCHAR(50) NOT NULL UNIQUE,
    DefectPercent    DECIMAL(5,2) NOT NULL CHECK (DefectPercent >= 0 AND DefectPercent < 100)
);
GO

-- Продукция
CREATE TABLE Products (
    ProductId       INT           NOT NULL IDENTITY(1,1) PRIMARY KEY,
    Article         NVARCHAR(50)  NULL,
    ProductTypeId   INT           NULL,
    ProductName     NVARCHAR(300) NOT NULL UNIQUE,
    MinPartnerPrice DECIMAL(10,2) NULL CHECK (MinPartnerPrice >= 0),
    CONSTRAINT FK_Products_ProductTypes FOREIGN KEY (ProductTypeId) REFERENCES ProductTypes(ProductTypeId)
);
GO

-- История реализации продукции партнером
CREATE TABLE PartnerProducts (
    PartnerProductId INT  NOT NULL IDENTITY(1,1) PRIMARY KEY,
    PartnerId        INT  NOT NULL,
    ProductId        INT  NOT NULL,
    Quantity         INT  NOT NULL CHECK (Quantity > 0),
    SaleDate         DATE NOT NULL,
    CONSTRAINT FK_PartnerProducts_Partners FOREIGN KEY (PartnerId) REFERENCES Partners(PartnerId),
    CONSTRAINT FK_PartnerProducts_Products FOREIGN KEY (ProductId) REFERENCES Products(ProductId)
);
GO

-- ПРОВЕРКА: скидки по партнерам (0/5/10/15%)
SELECT p.PartnerName,
       ISNULL(SUM(pp.Quantity), 0) AS TotalSold,
       CASE WHEN ISNULL(SUM(pp.Quantity), 0) < 10000 THEN 0
            WHEN SUM(pp.Quantity) < 50000            THEN 5
            WHEN SUM(pp.Quantity) <= 300000          THEN 10
            ELSE 15 END AS Discount
FROM Partners p
LEFT JOIN PartnerProducts pp ON pp.PartnerId = p.PartnerId
GROUP BY p.PartnerName;
GO
